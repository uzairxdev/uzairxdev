<!-- 🐍 SNAKE ANIMATION SETUP
     ─────────────────────────
     To enable the contribution snake, create this file in your profile repo:
     .github/workflows/snake.yml

     Paste this workflow:

     name: Generate Snake

     on:
       schedule:
         - cron: "0 0 * * *"
       workflow_dispatch:

     jobs:
       build:
         runs-on: ubuntu-latest
         permissions:
           contents: write
         steps:
           - uses: actions/checkout@v4

           - uses: Platane/snk@v3
             id: snake-gif
             with:
               github_user_name: ${{ github.repository_owner }}
               outputs: |
                 dist/github-snake.svg
                 dist/github-snake-dark.svg?palette=github-dark

           - uses: crazy-max/ghaction-github-pages@v4
             with:
               target_branch: output
               build_dir: dist
             env:
               GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

     Then go to Settings > Actions > General > Workflow permissions
     and enable "Read and write permissions".
     Run the workflow once manually from the Actions tab.
